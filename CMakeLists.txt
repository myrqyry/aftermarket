cmake_minimum_required(VERSION 3.21)
project(aftermarket VERSION 0.16.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  add_compile_options(/utf-8)

if(NOT AFTERMARKET_PACKAGE_LAYOUT MATCHES "^(legacy|modern)$")
  message(FATAL_ERROR "Unknown AFTERMARKET_PACKAGE_LAYOUT")
endif()

# Pinned, unmodified MIT libebur128. No network needed at build time.
add_library(aftermarket_ebur128 STATIC third_party/libebur128/ebur128/ebur128.c)
set_target_properties(aftermarket_ebur128 PROPERTIES POSITION_INDEPENDENT_CODE ON C_VISIBILITY_PRESET hidden)
target_include_directories(aftermarket_ebur128 PUBLIC third_party/libebur128/ebur128 PRIVATE third_party/libebur128/ebur128/queue)
if(MSVC)
  target_compile_definitions(aftermarket_ebur128 PRIVATE _USE_MATH_DEFINES)
else()
  target_link_libraries(aftermarket_ebur128 PRIVATE m)
endif()
add_library(mixer_loudness STATIC src/mixer-loudness.cpp)
set_target_properties(mixer_loudness PROPERTIES POSITION_INDEPENDENT_CODE ON)

target_include_directories(mixer_loudness PUBLIC src)

target_include_directories(augmentation PUBLIC src)
target_link_libraries(augmentation PUBLIC Qt6::Widgets Qt6::Svg)
# The native column adapter is intentionally limited to the inspected Qt
# private layout ABI. Other builds retain density and explicit overflow.
if(Qt6_VERSION VERSION_EQUAL "6.4.2" AND TARGET Qt6::WidgetsPrivate)
  target_link_libraries(augmentation PRIVATE Qt6::WidgetsPrivate)
  target_compile_definitions(augmentation PRIVATE AFTERMARKET_QT642_COLUMNS=1)
endif()
add_library(aftermarket MODULE src/plugin.cpp src/property-preset-footer.cpp src/filter-chain-presets.cpp src/obs-audio.cpp src/obs-transform.cpp src/obs-routing.cpp src/native-action-bridge.cpp src/command-center.cpp src/inspector-edits.cpp src/source-inspector-dock.cpp src/native-notes.cpp src/quick-find.cpp src/quick-find-obs.cpp src/source-dock-pin.cpp src/source-performance.cpp src/native-audio-settings.cpp src/obs-signal-levels.cpp src/obs-mixer-loudness.cpp)

set_target_properties(aftermarket PROPERTIES PREFIX "")
set(_notice "${CMAKE_CURRENT_SOURCE_DIR}/third_party/libebur128/THIRD_PARTY_NOTICES.txt")
if(APPLE)
  set_target_properties(aftermarket PROPERTIES BUNDLE TRUE BUNDLE_EXTENSION plugin
    MACOSX_BUNDLE_GUI_IDENTIFIER "com.myrqyry.aftermarket"
    MACOSX_BUNDLE_BUNDLE_VERSION "${PROJECT_VERSION}"
    MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}"
    MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/cmake/PluginInfo.plist.in"
    INSTALL_RPATH "@executable_path/../Frameworks" INSTALL_RPATH_USE_LINK_PATH FALSE)
  target_sources(aftermarket PRIVATE "${_notice}")
  set_source_files_properties("${_notice}" PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
  install(TARGETS aftermarket BUNDLE DESTINATION . LIBRARY DESTINATION .)
else()
  set(_module_destination bin/64bit)
  if(AFTERMARKET_PACKAGE_LAYOUT STREQUAL "modern")
    set(_module_destination .)
  endif()
  install(TARGETS aftermarket LIBRARY DESTINATION "${_module_destination}" RUNTIME DESTINATION "${_module_destination}")
  install(FILES "${_notice}" DESTINATION data)
endif()
