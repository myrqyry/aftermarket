cmake_minimum_required(VERSION 3.21)
project(obs-ui-augment VERSION 0.5.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Svg)
find_package(libobs CONFIG REQUIRED)
find_package(obs-frontend-api CONFIG REQUIRED)
add_library(augmentation STATIC src/augment.cpp src/inspector.cpp src/gradient.cpp)
set_target_properties(augmentation PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(augmentation PUBLIC src)
target_link_libraries(augmentation PUBLIC Qt6::Widgets Qt6::Svg)
add_library(obs-ui-augment MODULE src/plugin.cpp src/obs-audio.cpp)
target_link_libraries(obs-ui-augment PRIVATE augmentation OBS::libobs OBS::obs-frontend-api)
set_target_properties(obs-ui-augment PROPERTIES PREFIX "")
include(CTest)
if(BUILD_TESTING)
  add_executable(audio-effect-test tests/audio-effect-test.cpp)
  set_target_properties(audio-effect-test PROPERTIES AUTOMOC ON)
  target_link_libraries(audio-effect-test PRIVATE augmentation)
  add_test(NAME audio-effect COMMAND audio-effect-test)
  set_tests_properties(audio-effect PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_executable(gradient-test tests/gradient-test.cpp)
  target_link_libraries(gradient-test PRIVATE augmentation)
  add_test(NAME gradient COMMAND gradient-test)
  set_tests_properties(gradient PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_executable(motion-test tests/motion-test.cpp)
  target_link_libraries(motion-test PRIVATE augmentation)
  add_test(NAME motion COMMAND motion-test)
  set_tests_properties(motion PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_executable(inspector-test tests/inspector-test.cpp)
  target_link_libraries(inspector-test PRIVATE augmentation)
  add_test(NAME inspector COMMAND inspector-test)
  set_tests_properties(inspector PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_executable(menu-test tests/menu-test.cpp)
  target_link_libraries(menu-test PRIVATE augmentation)
  add_test(NAME menu-augmentation COMMAND menu-test)
  set_tests_properties(menu-augmentation PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
