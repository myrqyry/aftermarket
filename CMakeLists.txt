cmake_minimum_required(VERSION 3.21)
project(obs-ui-augment VERSION 0.16.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Svg)
find_package(libobs CONFIG REQUIRED)
find_package(obs-frontend-api CONFIG REQUIRED)
add_library(augmentation STATIC src/augment.cpp src/inspector.cpp src/gradient.cpp src/panels.cpp src/menu-layout.cpp src/theme-picker.cpp src/transform-visual.cpp src/property-sections.cpp src/dock-faces.cpp src/audio-patchbay.cpp src/row-stripes.cpp src/menu-bar-compact.cpp)
set_target_properties(augmentation PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(augmentation PUBLIC src)
target_link_libraries(augmentation PUBLIC Qt6::Widgets Qt6::Svg)
# The native column adapter is intentionally limited to the inspected Qt
# private layout ABI. Other builds retain density and explicit overflow.
if(Qt6_VERSION VERSION_EQUAL "6.4.2" AND TARGET Qt6::WidgetsPrivate)
  target_link_libraries(augmentation PRIVATE Qt6::WidgetsPrivate)
  target_compile_definitions(augmentation PRIVATE AFTERMARKET_QT642_COLUMNS=1)
endif()
add_library(obs-ui-augment MODULE src/plugin.cpp src/obs-audio.cpp src/obs-transform.cpp src/obs-routing.cpp)
target_link_libraries(obs-ui-augment PRIVATE augmentation OBS::libobs OBS::obs-frontend-api)
set_target_properties(obs-ui-augment PROPERTIES PREFIX "")
include(CTest)
