cmake_minimum_required(VERSION 3.21)
project(obs-ui-augment VERSION 0.14.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Svg)
find_package(libobs CONFIG REQUIRED)
find_package(obs-frontend-api CONFIG REQUIRED)
add_library(augmentation STATIC src/augment.cpp src/inspector.cpp src/gradient.cpp src/panels.cpp src/menu-layout.cpp src/theme-picker.cpp)
set_target_properties(augmentation PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(augmentation PUBLIC src)
target_link_libraries(augmentation PUBLIC Qt6::Widgets Qt6::Svg)
# The native column adapter is intentionally limited to the inspected Qt
# private layout ABI. Other builds retain density and explicit overflow.
if(Qt6_VERSION VERSION_EQUAL "6.4.2" AND TARGET Qt6::WidgetsPrivate)
  target_link_libraries(augmentation PRIVATE Qt6::WidgetsPrivate)
  target_compile_definitions(augmentation PRIVATE AFTERMARKET_QT642_COLUMNS=1)
endif()
add_library(obs-ui-augment MODULE src/plugin.cpp src/obs-audio.cpp)
target_link_libraries(obs-ui-augment PRIVATE augmentation OBS::libobs OBS::obs-frontend-api)
set_target_properties(obs-ui-augment PROPERTIES PREFIX "")
include(CTest)
if(BUILD_TESTING)

  set_target_properties(knob-meter-test PROPERTIES AUTOMOC ON)
  target_link_libraries(knob-meter-test PRIVATE augmentation)
  add_test(NAME knob-meter COMMAND knob-meter-test)
  set_tests_properties(knob-meter PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")

  target_link_libraries(panels-test PRIVATE augmentation)

  set_target_properties(knob-test PROPERTIES AUTOMOC ON)
  target_link_libraries(knob-test PRIVATE augmentation)
  add_test(NAME knob COMMAND knob-test)
  set_tests_properties(knob PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_executable(audio-effect-test tests/audio-effect-test.cpp)
  set_target_properties(audio-effect-test PROPERTIES AUTOMOC ON)
  target_link_libraries(audio-effect-test PRIVATE augmentation)
